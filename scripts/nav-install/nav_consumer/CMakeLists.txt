cmake_minimum_required(VERSION 3.8)
project(nav_install_consumer)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
if(NOT DEFINED EXPECTED_PACKAGE_VERSION OR NOT DEFINED EXPECTED_CONTRACT)
  message(FATAL_ERROR "Explicit expected package and contract versions required")
endif()
find_package(openamr_nav_msgs ${EXPECTED_PACKAGE_VERSION} EXACT REQUIRED)
add_executable(smoke smoke.cpp)
target_compile_definitions(smoke PRIVATE EXPECTED_CONTRACT=${EXPECTED_CONTRACT})
ament_target_dependencies(smoke rclcpp openamr_nav_msgs)
install(TARGETS smoke DESTINATION lib/${PROJECT_NAME})
ament_package()
